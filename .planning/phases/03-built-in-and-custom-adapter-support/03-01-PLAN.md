```typescript
/**
 * Map CLI flags to js-debug native config fields per D-07 hybrid approach.
 * Preserves js-debug's native config structure while providing flag convenience.
 */
export function mapJsDebugFlags(flags: Record<string, unknown>): Record<string, unknown> {
  const mapped: Record<string, unknown> = {};
  
  // Common flags mapped to js-debug native fields
  if (flags.program !== undefined) mapped.program = flags.program;
  if (flags.cwd !== undefined) mapped.cwd = flags.cwd;
  if (flags.runtimeExecutable !== undefined) mapped.runtimeExecutable = flags.runtimeExecutable;
  if (flags.runtimeArgs !== undefined) mapped.runtimeArgs = flags.runtimeArgs;
  if (flags.url !== undefined) mapped.url = flags.url;
  if (flags.port !== undefined) mapped.port = flags.port;
  if (flags.args !== undefined) mapped.args = flags.args;
  if (flags.env !== undefined) mapped.env = flags.env;
  if (flags.sourceMaps !== undefined) mapped.sourceMaps = flags.sourceMaps;
  if (flags.outFiles !== undefined) mapped.outFiles = flags.outFiles;
  
  // Electron-specific flags
  if (flags.runtimeExecutable === 'electron') {
    // Electron main process debugging uses node type with electron binary
    mapped.type = 'node';
  }
  
  return mapped;
}

/**
 * Map CLI flags to debugpy native config fields.
 */
export function mapDebugpyFlags(flags: Record<string, unknown>): Record<string, unknown> {
  const mapped: Record<string, unknown> = {};
  
  if (flags.program !== undefined) mapped.program = flags.program;
  if (flags.cwd !== undefined) mapped.cwd = flags.cwd;
  if (flags.args !== undefined) mapped.args = flags.args;
  if (flags.env !== undefined) mapped.env = flags.env;
  if (flags.python !== undefined) mapped.python = flags.python;
  
  return mapped;
}
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| CLI flags → adapter config | User-provided adapter IDs and config flags are untrusted input |
| Config file → adapter registry | Custom adapter config JSON is untrusted external data |
| .vscode/launch.json → launch config | Third-party launch.json files may contain unexpected or malicious paths/commands |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-01 | Tampering | Custom adapter config | mitigate | Validate all custom adapter descriptors with Zod schema - reject configs with invalid command paths, args, or env vars |
| T-03-02 | Information Disclosure | Adapter config paths | mitigate | Resolve all paths relative to controlled roots (dapCliHome, cwd) - do not allow arbitrary filesystem paths in config |
| T-03-03 | Denial of Service | Launch config loading | mitigate | Limit .vscode/launch.json file size (256KB max), timeout file reads (1s), reject configs with > 100 entries |
| T-03-04 | Elevation of Privilege | Adapter command execution | accept | Users explicitly configure custom adapters - command execution is intended behavior. Document security considerations in user docs (Phase 4). |
| T-03-05 | Spoofing | Built-in adapter override | mitigate | Built-in adapter IDs cannot be overridden by custom config - registry checks built-ins first, custom adapters must use unique IDs |
</threat_model>
