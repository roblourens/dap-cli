```ts
import { createServer } from 'node:http';
import { createReadStream, statSync } from 'node:fs';

export async function startFakeReleaseServer(handlers: readonly FakeReleaseHandler[]): Promise<FakeReleaseServer> {
  let hitCount = 0;
  const hitsByPath = new Map<string, number>();
  const server = createServer((req, res) => {
    hitCount++;
    hitsByPath.set(req.url ?? '', (hitsByPath.get(req.url ?? '') ?? 0) + 1);
    const handler = handlers.find(h => h.match(req));
    if (!handler) { res.writeHead(404); res.end(); return; }
    Promise.resolve(handler.handle(req, res)).catch(err => {
      if (!res.writableEnded) { res.writeHead(500); res.end(String(err)); }
    });
  });
  await new Promise<void>(r => server.listen(0, '127.0.0.1', r));
  const addr = server.address(); // AddressInfo
  return {
    url: `http://127.0.0.1:${typeof addr === 'object' && addr ? addr.port : 0}`,
    get hitCount() { return hitCount; },
    get hitsByPath() { return hitsByPath; },
    close: () => new Promise<void>(r => server.close(() => r())),
  };
}

export function jsDebugTarballHandler({ version, archivePath }: { version: string; archivePath: string }): FakeReleaseHandler {
  const expectedPath = `/microsoft/vscode-js-debug/releases/download/v${version}/js-debug-dap-v${version}.tar.gz`;
  return {
    match: req => req.url === expectedPath,
    handle: (_req, res) => {
      const size = statSync(archivePath).size;
      res.writeHead(200, { 'Content-Type': 'application/octet-stream', 'Content-Length': String(size) });
      createReadStream(archivePath).pipe(res);
    },
  };
}
// delveArchiveHandler / fixedStatusHandler analogous.
```
